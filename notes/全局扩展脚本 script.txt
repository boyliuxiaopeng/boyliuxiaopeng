// 优化版；用于 Clash Verge Rev 全局扩展脚本（JavaScript）。
// DNS 设置统一由 DNS 覆写管理；本脚本仅重建 DNS 专用策略组。
// DNS-自动只在美国、日本、台湾节点中测速选择；各地区组可手选。
// 自动测速仅衡量测试网址连通性，不代表实际 DoH 查询延迟。
function main(config) {
  if (!config || typeof config !== "object" || Array.isArray(config)) {
    throw new Error("全局扩展脚本需要一个有效的配置对象");
  }
  // DNS-PROXY 等为本脚本保留组名；普通节点重名时明确报错。
  if (!Array.isArray(config["proxy-groups"])) {
    config["proxy-groups"] = [];
  }

  // 两位地区代码必须由常见分隔符或字符串边界包围。
  function codePattern(code) {
    return "(^|[\\s_\\-\\[\\]()|/.,:：·])" +
      code +
      "([\\s_\\-\\[\\]()|/.,:：·]|$)";
  }

  const regions = [
    {
      name: "DNS-美国",
      pattern:
        "🇺🇸|美国|美國|USA|United States|America|" +
        "Los Angeles|San Jose|San Francisco|Seattle|" +
        "Chicago|Dallas|Ashburn|Phoenix|New York|Washington|" +
        "洛杉矶|洛杉磯|旧金山|舊金山|西雅图|西雅圖|" +
        "芝加哥|达拉斯|達拉斯|纽约|紐約|华盛顿|華盛頓|" +
        codePattern("US")
    },
    {
      name: "DNS-日本",
      pattern:
        "🇯🇵|日本|Japan|Tokyo|Osaka|东京|東京|大阪|" +
        codePattern("JP")
    },
    {
      name: "DNS-台湾",
      pattern:
        "🇹🇼|台湾|台灣|臺灣|Taiwan|Taipei|Kaohsiung|Taichung|Tainan|" +
        "台北|臺北|高雄|台中|臺中|台南|臺南|" +
        codePattern("TW")
    },
    {
      name: "DNS-香港",
      pattern:
        "🇭🇰|香港|Hong Kong|HongKong|Kowloon|九龙|九龍|" +
        codePattern("HK")
    },
    {
      name: "DNS-新加坡",
      pattern:
        "🇸🇬|新加坡|狮城|獅城|Singapore|" +
        codePattern("SG") + "|" + codePattern("SGP")
    }
  ];

  const managedNames = [
    "DNS-PROXY",
    "DNS-自动",
    "DNS-全部",
    ...regions.map(region => region.name)
  ];

  const nodes = Array.isArray(config.proxies) ? config.proxies : [];
  nodes.forEach(node => {
    if (node && managedNames.includes(node.name)) {
      throw new Error("节点名称与 DNS 策略组冲突：" + node.name + "，请先重命名该节点");
    }
  });

  // 只删除本脚本管理的同名组，保留订阅其他组。
  config["proxy-groups"] = config["proxy-groups"].filter(
    group => group && !managedNames.includes(group.name)
  );

  const regionGroups = regions.map(region => ({
    name: region.name,
    type: "select",
    "include-all": true,
    "exclude-type": "Direct|Reject|RejectDrop|Compatible|Pass|Dns",
    "empty-fallback": "REJECT",
    filter: "(?i)(" + region.pattern + ")"
  }));

  const autoGroup = {
    name: "DNS-自动",
    type: "url-test",
    "include-all": true,
    "exclude-type": "Direct|Reject|RejectDrop|Compatible|Pass|Dns",
    "empty-fallback": "REJECT",
    // 自动选择尊重美、日、台偏好；香港、新加坡和全部节点由用户手选。
    filter:
      "(?i)(" +
      regions.slice(0, 3).map(region => region.pattern).join("|") +
      ")",
    // 组内没有节点时使用 REJECT，不静默改走直连。
    url: "https://www.gstatic.com/generate_204",
    "expected-status": 204,
    interval: 300,
    tolerance: 50,
    timeout: 5000,
    lazy: true
  };

  const allGroup = {
    name: "DNS-全部",
    type: "select",
    "include-all": true,
    "exclude-type": "Direct|Reject|RejectDrop|Compatible|Pass|Dns",
    "empty-fallback": "REJECT"
  };

  const proxyGroup = {
    name: "DNS-PROXY",
    type: "select",
    proxies: [
      "DNS-自动",
      ...regions.map(region => region.name),
      "DNS-全部"
    ]
  };

  config["proxy-groups"].push(
    ...regionGroups,
    autoGroup,
    allGroup,
    proxyGroup
  );

  return config;
}
