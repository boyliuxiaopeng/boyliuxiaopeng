function main(config) {
  if (!Array.isArray(config["proxy-groups"])) {
    config["proxy-groups"] = [];
  }

  // 两位地区代码必须由常见分隔符或字符串边界包围。
  function codePattern(code) {
    return "(^|[\\s_\\-\\[\\]()|/])" +
      code +
      "([\\s_\\-\\[\\]()|/]|$)";
  }

  const regions = [
    {
      name: "DNS-美国",
      pattern:
        "🇺🇸|美国|美國|USA|United States|America|" +
        "Los Angeles|San Jose|San Francisco|Seattle|" +
        "Chicago|Dallas|Ashburn|Phoenix|New York|Washington|" +
        "洛杉矶|洛杉磯|旧金山|舊金山|西雅图|西雅圖|" +
        "芝加哥|达拉斯|達拉斯|纽约|紐約|华盛顿|華盛頓|" +
        codePattern("US")
    },
    {
      name: "DNS-日本",
      pattern:
        "🇯🇵|日本|Japan|Tokyo|Osaka|东京|東京|大阪|" +
        codePattern("JP")
    },
    {
      name: "DNS-台湾",
      pattern:
        "🇹🇼|台湾|台灣|Taiwan|Taipei|Kaohsiung|Taichung|Tainan|" +
        "台北|臺北|高雄|台中|臺中|台南|臺南|" +
        codePattern("TW")
    },
    {
      name: "DNS-香港",
      pattern:
        "🇭🇰|香港|Hong Kong|HongKong|Kowloon|九龙|九龍|" +
        codePattern("HK")
    },
    {
      name: "DNS-新加坡",
      pattern:
        "🇸🇬|新加坡|狮城|獅城|Singapore|" +
        codePattern("SG") + "|" + codePattern("SGP")
    }
  ];

  const managedNames = [
    "DNS-PROXY",
    "DNS-自动",
    "DNS-全部",
    ...regions.map(region => region.name)
  ];

  // 只删除本脚本管理的同名组，保留订阅其他组。
  config["proxy-groups"] = config["proxy-groups"].filter(
    group => group && !managedNames.includes(group.name)
  );

  const regionGroups = regions.map(region => ({
    name: region.name,
    type: "select",
    "include-all": true,
    "exclude-type": "Direct|Reject",
    "empty-fallback": "REJECT",
    filter: "(?i)(" + region.pattern + ")"
  }));

  const autoGroup = {
    name: "DNS-自动",
    type: "url-test",
    "include-all": true,
    "exclude-type": "Direct|Reject",
    "empty-fallback": "REJECT",
    filter:
      "(?i)(" +
      regions.slice(0, 3).map(region => region.pattern).join("|") +
      ")",
    url: "https://www.gstatic.com/generate_204",
    "expected-status": 204,
    interval: 300,
    tolerance: 50,
    lazy: true
  };

  const allGroup = {
    name: "DNS-全部",
    type: "select",
    "include-all": true,
    "exclude-type": "Direct|Reject",
    "empty-fallback": "REJECT"
  };

  const proxyGroup = {
    name: "DNS-PROXY",
    type: "select",
    proxies: [
      "DNS-自动",
      ...regions.map(region => region.name),
      "DNS-全部"
    ]
  };

  config["proxy-groups"].push(
    ...regionGroups,
    autoGroup,
    allGroup,
    proxyGroup
  );

  return config;
}
